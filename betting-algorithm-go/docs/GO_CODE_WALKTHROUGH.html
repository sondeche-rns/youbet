predictMatch handler — anonymous struct + optional pointers

func (s *Server) predictMatch(w http.ResponseWriter, r *http.Request) {

    // Anonymous struct — inline type defined only for this request body
    var body struct {
        HomeTeam    string   `json:"home_team"`
        AwayTeam    string   `json:"away_team"`
        HomeOdds    *float64 `json:"home_odds"`  // pointer = optional field
        DrawOdds    *float64 `json:"draw_odds"`  // nil if not in request
        AwayOdds    *float64 `json:"away_odds"`
    }

    // Streaming JSON decode — reads directly from HTTP body, no intermediate buffer
    if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
        writeError(w, http.StatusBadRequest, "invalid JSON: "+err.Error())
        return
    }

    md := &algorithm.MatchData{
        HomeTeam: body.HomeTeam,
        AwayTeam: body.AwayTeam,
        HomeOdds: body.HomeOdds,  // pass pointer through — nil if absent
    }

    pred, err := s.engine.Predict(md)  // → Step 3
    if err != nil {
        writeError(w, http.StatusInternalServerError, err.Error())
        return
    }
    writeJSON(w, http.StatusOK, pred)
}
*float64 for optional values — Go has no null, no Optional<T>. A pointer to a type is Go's idiomatic optional: nil = absent, non-nil pointer = present. The JSON decoder sets pointer fields to nil when the key is missing from the JSON.
Anonymous struct — var body struct{ ... } defines a type inline, used only in this function scope. Avoids creating a named type for one-time use. Common for HTTP request bodies.

func (e *PredictionEngine) Predict(data *MatchData) (*PredictionResult, error) {

    e.setDefaults(data)                        // ① fill missing fields
    factors     := e.calculateAllFactors(data) // ② 16 factor calculations → Step 4
    rawProbs    := e.calculateWeightedProbabilities(factors)  // ③ → Step 6

    // ④ optional: blend with market odds (only when caller provided them)
    if data.HomeOdds != nil && data.DrawOdds != nil && data.AwayOdds != nil {
        marketProbs := oddsToProbs(*data.HomeOdds, *data.DrawOdds, *data.AwayOdds)
        quality     := e.assessDataQuality(data)          // 0.0 – 1.0
        modelWeight := 0.40 + quality*0.30               // 40-70% model, rest market
        rawProbs     = blendProbabilities(rawProbs, marketProbs, modelWeight)
    }

    // ⑤ Poisson blend (football only) → Step 7
    var blendedProbs Probabilities
    if e.sport == "football" {
        poissonProbs := e.calculatePoissonProbabilities(data, factors)
        blendedProbs  = blendProbabilities(rawProbs, poissonProbs, 0.6)
    } else {
        blendedProbs = rawProbs
    }

    calibratedProbs := e.calibrateProbabilities(blendedProbs) // ⑥ Platt scaling → Step 8
    confidence      := e.calculateConfidence(factors, calibratedProbs)
    recommendation  := e.generateRecommendation(calibratedProbs, confidence,
                         data.HomeOdds, data.DrawOdds, data.AwayOdds) // ⑦ Kelly → Step 9

    return &PredictionResult{
        HomeWinProb: round(calibratedProbs.Home, 4),
        DrawProb:    round(calibratedProbs.Draw, 4),
        AwayWinProb: round(calibratedProbs.Away, 4),
        Confidence:  round(confidence, 4),
        Recommendation: recommendation,
        Timestamp:   time.Now().Format(time.RFC3339),
    }, nil
}
Dereferencing *data.HomeOdds — Since HomeOdds is *float64, you must nil-check it first, then dereference with * to get the underlying float64. The nil guard and dereference always appear together.

formToScore — rune iteration + comma-ok map lookup

func formToScore(form string) float64 {
    // Map literal: rune keys, float64 values
    points  := map[rune]float64{'W': 1.0, 'D': 0.5, 'L': 0.0}
    weights := []float64{0.3, 0.25, 0.2, 0.15, 0.1}  // recent result = higher weight
    score   := 0.0

    // Range over string yields runes (Unicode code points) not bytes
    for i, result := range []rune(form) {
        if i >= 5 { break }
        weight := 0.1
        if i < len(weights) { weight = weights[i] }

        // Comma-ok map lookup — never panics on missing key
        if val, ok := points[result]; ok {
            score += val * weight
        }
    }
    return score
}
Rune vs byte — Strings in Go are UTF-8 byte slices. Ranging over a string yields rune (Unicode code point), not raw bytes. For ASCII strings (W/D/L) this is the same, but iterating as runes is the correct general pattern.
Comma-ok map lookup val, ok := m[key] — If the key exists, ok is true. If absent, ok is false and val is the zero value. Never panics. Always prefer this over plain m[key] when the key may be absent.

Separating two factor types via type assertion

func (e *PredictionEngine) calculateWeightedProbabilities(factors map[string]interface{}) Probabilities {
    activeFactors  := make(map[string]factorData)
    drawBoostTotal := 0.0

    for name, raw := range factors {

        // Safe type assertion (comma-ok) — no panic on wrong type
        if fr, ok := raw.(domain.FactorResult); ok {
            if fr.Triggered && fr.Weight > 0 {
                activeFactors[name] = factorData{score: fr.Value, weight: fr.Weight}
                if val, ok := fr.Metadata["draw_boost"].(float64); ok {
                    drawBoostTotal += val
                }
            }
        } else if dict, ok := raw.(map[string]interface{}); ok {
            if score, ok := dict["score"].(float64); ok {
                if w := e.config.Weights[name]; w > 0 {
                    activeFactors[name] = factorData{score: score, weight: w}
                }
            }
        }
    }

    // Dynamic normalisation — only active factors contribute
    totalWeight := 0.0
    for _, fd := range activeFactors { totalWeight += fd.weight }

    weightedScore := 0.0
    for _, fd := range activeFactors {
        weightedScore += fd.score * (fd.weight / totalWeight)
    }
    // weightedScore in (0,1): >0.5 = home favoured, <0.5 = away favoured

    drawProb := calculateDrawProbability(weightedScore, factors, drawBoostTotal)
    homeProb := weightedScore * (1 - drawProb)
    awayProb := (1 - weightedScore) * (1 - drawProb)

    // Clamp and normalise to sum = 1.0
    homeProb = math.Max(0.05, math.Min(0.90, homeProb))
    drawProb = math.Max(0.05, math.Min(0.90, drawProb))
    awayProb = math.Max(0.05, math.Min(0.90, awayProb))
    total    := homeProb + drawProb + awayProb

    return Probabilities{Home: homeProb/total, Draw: drawProb/total, Away: awayProb/total}
}
Type assertion v, ok := x.(T) — The safe form. If x holds a value of type T, ok is true. If not, ok is false and v is the zero value. The unsafe form v := x.(T) panics on wrong type — avoid it.
Map iteration order is random in Go — Iterating a map with for k, v := range m does not guarantee any order. This is intentional — Go deliberately randomizes it to prevent relying on accidental ordering. If order matters, sort the keys first.

import "gonum.org/v1/gonum/stat/distuv"   // Go's NumPy/SciPy equivalent

// Create two independent Poisson distributions — one per team
homePoisson := distuv.Poisson{Lambda: homeXg}   // λ = expected goals
awayPoisson := distuv.Poisson{Lambda: awayXg}

homeWin, draw, awayWin := 0.0, 0.0, 0.0

// Iterate all 36 scorelines 0-0 through 5-5
for h := 0; h <= 5; h++ {
    for a := 0; a <= 5; a++ {
        // P(scoreline h:a) = P(home scores h) × P(away scores a)
        prob := homePoisson.Prob(float64(h)) * awayPoisson.Prob(float64(a))
        switch {
        case h > a:  homeWin += prob
        case h == a: draw    += prob
        default:    awayWin += prob
        }
    }
}
total := homeWin + draw + awayWin
return Probabilities{Home: homeWin/total, Draw: draw/total, Away: awayWin/total}
Why Poisson for football? Goals are rare, independent events — exactly the Poisson distribution's domain. Each team's goals are modelled as an independent Poisson process with λ = expected goals (xG). Multiplying the two PMFs gives the probability of any exact scoreline.
Why stop at 5-5? Beyond 5 goals per team, probabilities are negligible (<0.01%). The 36 scorelines 0-0 through 5-5 cover 99%+ of total probability mass, making the approximation effectively exact.

sync.RWMutex — Allows multiple concurrent readers OR one exclusive writer, never both. Use RLock()/RUnlock() for read-only operations, Lock()/Unlock() for mutations. More efficient than a plain Mutex when reads are much more frequent than writes.
defer — Schedules a call to run when the surrounding function returns — whether normally, from an early return, or even via panic. Used here to guarantee the mutex is always unlocked. Without defer, a panic would deadlock every goroutine waiting on the lock.

Defensive map copy

// WRONG — both variables point to the SAME underlying map
copy := currentWeights   // mutations to copy affect currentWeights too!

// CORRECT — manually copy each entry into a new map
func copyWeights(weights map[string]float64) map[string]float64 {
    result := make(map[string]float64, len(weights))
    for k, v := range weights { result[k] = v }
    return result
}
Maps are reference types. Assigning a map variable copies only the reference — both variables point to the same data. To create an independent copy, iterate and copy entries manually (or use maps.Clone() in Go 1.21+).

Finding the best bet — nil pointer to anonymous struct

func (e *PredictionEngine) generateRecommendation(
    probs Probabilities, confidence float64,
    homeOdds, drawOdds, awayOdds *float64,
) *Recommendation {

    bestEV := 0.0
    // Pointer to anonymous struct — starts nil, only set if a positive-EV bet found
    var bestBet *struct{ outcome string; prob float64; odds float64 }

    if homeOdds != nil {
        ev := (probs.Home * *homeOdds) - 1   // EV = (probability × odds) - 1
        if ev > bestEV {
            bestEV  = ev
            bestBet = &struct{ outcome string; prob float64; odds float64 }{
                "Home Win", probs.Home, *homeOdds,
            }
        }
    }
    // Same for draw and away ...

    // Kelly Criterion
    if bestBet != nil && bestEV > 0 {
        b      := bestBet.odds - 1   // net odds (profit per unit staked)
        kelly  := (b*bestBet.prob - (1-bestBet.prob)) / b
        // Apply fractional Kelly (25%) and cap at 5% max stake
        stakePct = math.Min(0.05, math.Max(0, kelly*0.25))
    }

    switch {
    case bestEV > 0.10 && confidence > 0.70: recType = "Strong Bet"
    case bestEV > 0.05 && confidence > 0.60: recType = "Value Bet"
    case bestEV > 0:                           recType = "Small Edge"
    default:                                   recType = "No Bet"
    }
    // ...
}
Kelly Criterion — the formula: f* = (b×p - q) / b where b = odds−1 (profit per unit), p = win probability, q = 1−p (loss probability). Fractional Kelly (×0.25) and a 5% cap protect the bankroll from ruin in volatile sequences.

func (s *Server) startDataCollection(w http.ResponseWriter, r *http.Request) {

    // ① Check current state (read lock — safe for concurrent status polls)
    s.collectMu.RLock()

    s.collectStatus = data.CollectionStatus{Running: true, Step: "Starting..."}
    s.collectMu.Unlock()

    // ③ Launch goroutine — this line returns immediately
    go func() {
        // This closure captures s (the server pointer) from the outer scope
        callback := func(step string, current, total int) {
            s.collectMu.Lock()
            s.collectStatus.Step     = step
            s.collectStatus.Progress = int(float64(current)/float64(total)*100)
            s.collectMu.Unlock()
        }

        // 7-step pipeline runs here in the goroutine
        _, err := s.collector.CollectAllData(seasons, leagues, callback)

        // ④ Update completion state
        s.collectMu.Lock()
        s.collectStatus.Running   = false
        s.collectStatus.Completed = err == nil
        if err != nil { s.collectStatus.Error = err.Error() }
        s.collectMu.Unlock()
    }()  // ← () invokes the goroutine function immediately

    // ⑤ HTTP response returns here — before the pipeline finishes
    writeJSON(w, http.StatusOK, map[string]any{"message": "Data collection started"})
}
go func() { ... }() — The go keyword launches the function in a new goroutine. A goroutine is a lightweight thread managed by the Go runtime — starts in microseconds (not milliseconds), uses 2KB initial stack (not 1MB). The () at the end immediately calls the anonymous function in the new goroutine.
Closure captures s — The goroutine is a closure that captures the variable s (the server pointer) from the surrounding scope. Both the goroutine and the handler share the same s, which is why mutex protection is required for every access to s.collectStatus.
